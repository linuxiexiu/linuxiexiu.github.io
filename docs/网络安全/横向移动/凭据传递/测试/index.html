        • 终极修复版脚本
        • 为什么之前的脚本会“马上没有”？
        • 如何使用修复版
        • 测试步骤

    这个问题出现的核心原因是：脚本在执行完 case 中的命令后，立即执行了 read -p "按回车返回菜单..."，但在此之前，屏幕可能因为子进程的输出、日志打印或者终端刷新机制被“冲掉”了，或者脚本因为权限/环境问题在后台异常退出了。

    特别是你之前的脚本中使用了 su -c 和复杂的管道，这在 Termux 的 sh 中极易导致脚本本身意外终止，或者 read 命令无法正确读取输入（因为标准输入流被前面的命令占用了）。

    终极修复版脚本#

    这个版本做了以下关键修改来解决“闪退”和“输入无效”的问题：

    1. 移除 su：彻底删除 su 调用，避免权限错误导致脚本崩溃。
    2. 强制刷新与暂停：在关键操作后增加 sleep 和明确的 echo 提示，确保你有时间看到输出。
    3. 输入流保护：优化 read 命令，防止被后台日志干扰。
    4. 环境加载前置：在脚本一开始就加载一次环境变量，而不是每次都在子 shell 里加载，提高稳定性。
    5. 错误捕获：如果命令执行失败，明确打印错误信息而不是直接消失。

    请替换你原来的脚本内容：

    #!/data/data/com.termux/files/usr/bin/bash

# ==========================================
# Hugo 服务管理中心 (修复闪退版)
# ==========================================

# 配置
HUGO_PROJECT_DIR="${HOME}"
HUGO_CMD="hugo server --noBuildLock --disableFastRender --bind 0.0.0.0 --port 1313"
LOG_FILE="${HOME}/hugo_server.log"
PID_FILE="${HOME}/hugo_server.pid"
ENV_FILE="${HOME}/termux-env"

# 颜色
GREEN="\033[32m"
RED="\033[31m"
YELLOW="\033[33m"
BLUE="\033[34m"
RESET="\033[0m"

# 确保环境变量已加载 (修复找不到 hugo 命令的问题)
if [ -f "$ENV_FILE" ]; then
    . "$ENV_FILE"
elif [ -f "$HOME/.bashrc" ]; then
    . "$HOME/.bashrc"
fi

# 获取 IP
get_local_ip() {
    local ip=$(ifconfig wlan0 2>/dev/null | grep 'inet ' | awk '{print $2}' | cut -d: -f2)
    [ -z "$ip" ] && ip=$(ip addr show wlan0 2>/dev/null | grep 'inet ' | awk '{print $2}' | cut -d/ -f1)
    [ -z "$ip" ] && ip="127.0.0.1"
    echo "$ip"
}

# 检查状态
is_running() {
    if [ -f "$PID_FILE" ]; then
        local pid=$(cat "$PID_FILE" 2>/dev/null)
        if [ -n "$pid" ] && kill -0 "$pid" 2>/dev/null; then
            # 额外检查是否是 hugo 进程
            if ps -o command= -p "$pid" 2>/dev/null | grep -q "hugo"; then
                return 0
            fi
        fi
    fi
    return 1
}

# 暂停函数 (解决输入被跳过的问题)
pause_menu() {
    echo ""
    echo -e "${BLUE}----------------------------------------${RESET}"
    echo -e "${YELLOW}按 [回车键] 返回主菜单...${RESET}"
    echo -e "${BLUE}----------------------------------------${RESET}"
    read -r dummy_input
}

# 主循环
while true; do
    clear
    echo -e "${BLUE}========================================${RESET}"
    echo -e "      ${GREEN}Hugo 服务 后台管理中心${RESET}"
    echo -e "${BLUE}========================================${RESET}"
    
    if is_running; then
        local pid=$(cat "$PID_FILE")
        echo -e "🔧 状态: ${GREEN}✅ 运行中${RESET} (PID: ${pid})"
        echo -e "🌐 地址: ${YELLOW}http://$(get_local_ip):1313${RESET}"
    else
        echo -e "🔧 状态: ${RED}❌ 已停止${RESET}"
        rm -f "$PID_FILE" 2>/dev/null
    fi
    
    echo -e "📂 路径: $HUGO_PROJECT_DIR"
    echo -e "${BLUE}========================================${RESET}"
    echo "1) 启动服务"
    echo "2) 重启服务"
    echo "3) 停止服务"
    echo "4) 查看日志 (Ctrl+C 退出)"
    echo "5) 测试环境 (检查 Hugo)"
    echo "6) 退出脚本"
    echo -e "${BLUE}========================================${RESET}"
    
    # 使用 -r 防止转义字符干扰
    read -r -p "请输入编号 [1-6]: " choice

    case $choice in
        1)
            if is_running; then
                echo -e "\n${YELLOW}⚠️  服务已在运行，无需重复启动！${RESET}"
            else
                echo -e "\n${BLUE}正在启动...${RESET}"
                cd "$HUGO_PROJECT_DIR" || { echo -e "${RED}目录不存在！${RESET}"; pause_menu; continue; }
                
                # 后台启动，重定向输出
                nohup $HUGO_CMD > "$LOG_FILE" 2>&1 &
                local new_pid=$!
                echo "$new_pid" > "$PID_FILE"
                
                sleep 2 # 等待启动
                
                if is_running; then
                    echo -e "${GREEN}✅ 启动成功！${RESET}"
                    echo -e "🌐 访问: http://$(get_local_ip):1313"
                else
                    echo -e "${RED}❌ 启动失败！${RESET}"
                    echo -e "最后 5 行错误日志:"
                    tail -n 5 "$LOG_FILE"
                fi
            fi
            pause_menu
            ;;

        2)
            echo -e "\n${BLUE}正在重启...${RESET}"
            if is_running; then
                kill "$(cat "$PID_FILE")" 2>/dev/null
                sleep 1
                rm -f "$PID_FILE"
            fi
            
            cd "$HUGO_PROJECT_DIR" || { echo -e "${RED}目录不存在！${RESET}"; pause_menu; continue; }
            nohup $HUGO_CMD > "$LOG_FILE" 2>&1 &
            echo $! > "$PID_FILE"
            
            sleep 2
            if is_running; then
                echo -e "${GREEN}✅ 重启完成！${RESET}"
            else
                echo -e "${RED}❌ 重启失败，请检查日志。${RESET}"
                tail -n 5 "$LOG_FILE"
            fi
            pause_menu
            ;;

        3)
            if is_running; then
                kill "$(cat "$PID_FILE")" 2>/dev/null
                sleep 1
                # 如果还在运行，强制杀
                if is_running; then kill -9 "$(cat "$PID_FILE")" 2>/dev/null; fi
                rm -f "$PID_FILE"
                echo -e "\n${GREEN}✅ 服务已停止！${RESET}"
            else
                echo -e "\n${YELLOW}⚠️  服务未运行。${RESET}"
            fi
            pause_menu
            ;;

        4)
            clear
            echo -e "${BLUE}========================================${RESET}"
            echo -e "     📄 实时日志 (按 Ctrl+C 返回菜单)"
            echo -e "${BLUE}========================================${RESET}"
            if [ -f "$LOG_FILE" ]; then
                tail -n 30 -f "$LOG_FILE"
            else
                echo -e "${YELLOW}日志文件暂无内容。${RESET}"
                sleep 2
            fi
            # tail -f 被 Ctrl+C 中断后，会继续执行下面的代码
            echo -e "\n${GREEN}已退出日志查看。${RESET}"
            pause_menu
            ;;

        5)
            echo -e "\n${BLUE}正在检查环境...${RESET}"
            if command -v hugo &> /dev/null; then
                echo -e "${GREEN}✅ Hugo 已找到！${RESET}"
                echo "版本信息:"
                hugo version
            else
                echo -e "${RED}❌ 未找到 Hugo 命令！${RESET}"
                echo "请确保已运行: pkg install hugo"
                echo "或者检查 $ENV_FILE 是否正确配置了 PATH。"
            fi
            pause_menu
            ;;

        6)
            echo -e "\n${GREEN}👋 脚本退出，服务将在后台继续运行。${RESET}"
            exit 0
            ;;

        *)
            echo -e "\n${RED}❌ 无效输入！请输入 1-6。${RESET}"
            sleep 1
            ;;
    esac
done

    为什么之前的脚本会“马上没有”？#

    1. su -c 导致崩溃：在 Termux 中，普通用户执行 su 通常会报错 Permission denied 或直接退出。一旦脚本报错退出，终端就会回到命令行提示符，看起来就像“内容闪过然后没了”。
    2. read 被跳过：如果前面的命令（如 su 或 nohup 的某些变体）占用了标准输入（stdin），或者脚本因为错误非正常退出，read -p 根本不会执行，或者执行了但立刻因为 EOF（文件结束）而返回。
    3. 清屏太快：之前的逻辑中，如果出错没有暂停，直接循环 clear，导致你看不到错误信息。

    1. 运行脚本。
    2. 输入 5 (测试环境)。如果这里能看到 Hugo 版本号，说明环境没问题。如果这里显示红色“未找到”，请先在终端手动运行 pkg install hugo。
    3. 输入 1 (启动)。
      • 现在它应该会显示“正在启动…”，等待 2 秒，然后显示“✅ 启动成功”或具体的错误日志。
      • 关键点：最后会显示 [按回车键 返回主菜单...]，并且光标会停在那里等你，不会闪退。
